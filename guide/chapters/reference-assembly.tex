\section{Common Rules}

Include \cmd{ultimate.inc} and link \cmd{ultimate.lib}. These are native 6502
entries: arguments use registers or the exported \cmd{ult\_*} cells, never the
C stack. Calls are not re-entrant. They install no interrupt handler and are
safe with interrupts disabled. The SDK owns four zero-page bytes at \$FB--\$FE
unless \cmd{UCI\_ZP} is changed before assembly.

Except where an entry returns a value, \cmd{A} is an \cmd{ULTIMATE\_*} result;
\cmd{beq} therefore branches on success. A 16-bit value returned in \cmd{A/X}
has no flag guarantee. Initialize UCI-backed services with
\cmd{ultimate\_init}; turbo, REU, audio and vsprites state their own probes.
Pointers are little-endian, with the low byte in \cmd{A} and high byte in
\cmd{X}. Strings sent to the Ultimate must be ASCII, not ca65-mapped PETSCII.
Examples reuse the ordinary \cmd{setptr}, \cmd{setword} and \cmd{setdword}
macros defined in Chapter~4.

\section{Initialization And Discovery}

\asmref{ultimate\_init}
{None.}
{\cmd{A} is \cmd{ULTIMATE\_OK}, \cmd{ULTIMATE\_ERR\_NO\_DEVICE}, or
\cmd{ULTIMATE\_ERR\_TIMEOUT}; N and Z reflect \cmd{A}.}
{}
{Initializes and functionally checks the command interface.}
{jsr ultimate\_init\
bne no\_ultimate}
{The branch is taken unless the SDK is ready.}

\asmref{ultimate\_available}
{None.}
{\cmd{A} is 1 after successful initialization, otherwise 0; N and Z reflect
\cmd{A}.}
{}
{Reads cached availability without touching hardware.}
{jsr ultimate\_available\
beq not\_ready}
{The branch is taken until \cmd{ultimate\_init} succeeds.}

\asmref{ultimate\_detect}
{\cmd{A/X} points to four writable bytes: present, ident and a 16-bit target
mask.}
{\cmd{A} is a result code.}
{\cmd{ult\_buf}, \cmd{ult\_buflen} and \cmd{ult\_outlen}.}
{Probes every known target and fills a reusable capability block.}
{lda \#<caps\
ldx \#>caps\
jsr ultimate\_detect}
{On success, bit N of \cmd{caps+2} identifies target N.}

\asmref{ultimate\_identify}
{\cmd{A} is a \cmd{UCI\_TARGET\_*}; \cmd{ult\_buf} points to output,
\cmd{ult\_buflen} is its capacity and \cmd{ult\_outlen} points to an optional
16-bit stored length.}
{\cmd{A} is a result code; the text is NUL-terminated when capacity permits.}
{}
{Reads one target's identification string.}
{lda \#UCI\_TARGET\_DOS1\
jsr ultimate\_identify}
{The caller's buffer contains the DOS identity on success.}

\asmref{ultimate\_get\_model}
{\cmd{ult\_buf}, \cmd{ult\_buflen} and optional \cmd{ult\_outlen} describe
the output.}
{\cmd{A} is a result code; output is terminated when capacity permits.}
{}
{Reads the model using deprecated control HWINFO compatibility.}
{jsr ultimate\_get\_model}
{Handle \cmd{ULTIMATE\_ERR\_NOT\_SUPPORTED}; HWINFO may disappear.}

\asmref{ultimate\_legacy\_get\_sid\_info}
{\cmd{A/X} points to \cmd{ULTIMATE\_SID\_INFO\_SIZE} writable bytes.}
{\cmd{A} is a result code; byte zero is the record count on success.}
{}
{Reads up to four raw SID mapping records through deprecated HWINFO.}
{lda \#<sidinfo\
ldx \#>sidinfo\
jsr ultimate\_legacy\_get\_sid\_info}
{Records begin at \cmd{ULTIMATE\_SID\_INFO\_RECORDS}; each five-byte record is
a little-endian primary address, little-endian secondary address, and raw type
byte at the named \cmd{ULTIMATE\_SID\_*} offsets.}

\asmref{ultimate\_strerror}
{\cmd{A} is an \cmd{ULTIMATE\_*} result code.}
{\cmd{A/X} points to a NUL-terminated PETSCII explanation.}
{}
{Maps an SDK result to printable text.}
{lda \#ULTIMATE\_ERR\_IO\
jsr ultimate\_strerror}
{\cmd{A/X} points to \cmd{I/O ERROR}.}

\asmref{ultimate\_turbo\_available}

\asmref{ultimate\_turbo\_get}
{None.}
{\cmd{A} is the current speed index or \cmd{U64\_TURBO\_UNAVAILABLE} (\$FF).}
{}

{jsr ultimate\_turbo\_get\

{\cmd{old\_speed} can restore a supported entry speed.}

\asmref{ultimate\_turbo\_set}
{\cmd{A} is a speed index 0--15; 1--4\,MHz constants are portable and

{\cmd{A} is a result code.}
{}
{Sets CPU speed and leaves badlines unchanged.}
{lda \#U64\_SPEED\_1MHZ:jsr ultimate\_turbo\_set\
lda \#U64\_SPEED\_2MHZ:jsr ultimate\_turbo\_set\
lda \#U64\_SPEED\_3MHZ:jsr ultimate\_turbo\_set\
lda \#U64\_SPEED\_4MHZ:jsr ultimate\_turbo\_set\
lda \#U64\_SPEED\_MAX:jsr ultimate\_turbo\_set}
{The calls demonstrate every named portable speed and model maximum.}

\asmref{ultimate\_turbo\_badlines}
{\cmd{A}=0 disables badlines; any nonzero value enables them.}
{\cmd{A} is a result code.}
{}
{Changes VIC cycle stealing without changing CPU speed.}
{lda \#0:jsr ultimate\_turbo\_badlines\
lda \#1:jsr ultimate\_turbo\_badlines}
{The two calls demonstrate both settings.}

\asmref{ultimate\_vsprite\_draw}
{\cmd{A/X} points to a \cmd{VSPRITE\_SIZE} descriptor. Its named offsets hold
bitmap, source, optional mask and screen pointers; byte-column \cmd{X} (0--39),
raster-line \cmd{Y} (0--199), nonzero width and height that fit the bitmap,
colour, and exactly one operation: 0, \cmd{VSPRITE\_F\_MASKED},
\cmd{VSPRITE\_F\_COPY}, or \cmd{VSPRITE\_F\_COLOR}. Image and mask bytes are
column-major.}
{\cmd{A} is a result code; the destination bitmap changes in place.}
{The routine self-modifies and therefore must execute from RAM.}
{Draws OR, masked, bitmap-copy or colour-aware software vsprites without UCI.}
{lda \#0:sta sprite+VSPRITE\_FLAGS\
lda \#<sprite:ldx \#>sprite:jsr ultimate\_vsprite\_draw\
lda \#VSPRITE\_F\_MASKED:sta sprite+VSPRITE\_FLAGS\
lda \#<sprite:ldx \#>sprite:jsr ultimate\_vsprite\_draw\
lda \#VSPRITE\_F\_COPY:sta sprite+VSPRITE\_FLAGS\
lda \#<sprite:ldx \#>sprite:jsr ultimate\_vsprite\_draw\
lda \#VSPRITE\_F\_COLOR:sta sprite+VSPRITE\_FLAGS\
lda \#<sprite:ldx \#>sprite:jsr ultimate\_vsprite\_draw}
{The four calls select OR, masked, bitmap-copy and colour-aware draws.}

\section{Loading And Saving Memory}

\asmref{ultimate\_load}
{\cmd{ult\_buf} points to a PRG name; \cmd{ult\_addr}=0 uses its header and a

{\cmd{A} is a result code; \cmd{ult\_end} is the address after the load.}
{The DOS fallback changes \cmd{ult\_buf}, \cmd{ult\_buflen},
\cmd{ult\_outlen}, \cmd{ult\_addr} and \cmd{ult\_max}.}
{Consumes the PRG header and tries SoftwareIEC before DOS fallback.}
{setptr ult\_buf,name\

jsr ultimate\_load}
{The PRG loads at its stored address on success.}

\asmref{ultimate\_bload}
{\cmd{ult\_buf} points to a name; nonzero \cmd{ult\_addr} is destination and

{\cmd{A} is a result code; \cmd{ult\_end} is the address after the load.}
{}
{Loads uninterpreted bytes without consuming a header.}
{setword ult\_addr,\$c000\
setword ult\_max,2048\
jsr ultimate\_bload}
{At most 2048 bytes are written from \$C000.}

\asmref{ultimate\_save}
{\cmd{ult\_buf} points to a name, \cmd{ult\_addr} is the C64 start and
nonzero \cmd{ult\_max} is the length.}
{\cmd{A} is a result code.}
{}
{Creates or replaces a file from C64 memory.}
{setword ult\_addr,\$0400\
setword ult\_max,1000\
jsr ultimate\_save}
{The text screen is saved on success.}

\asmref{ultimate\_last\_end}
{None.}
{\cmd{A/X} is the address after the last successful load.}
{}
{Reads cached load state without touching hardware.}
{jsr ultimate\_last\_end\
sta end\
stx end+1}
{\cmd{end-1} is the last address written.}

\section{RAM Expansion}

\asmref{ultimate\_reu\_available}
{None; UCI initialization is not required.}
{\cmd{A} is 1 when REU registers answer, otherwise 0.}
{}
{Probes for enabled RAM expansion.}
{jsr ultimate\_reu\_available\
beq no\_reu}
{The branch is taken when no REU is mapped.}

\asmref{ultimate\_reu\_size}
{None.}
{\cmd{A/X} is the expansion size in 64K banks, or zero when absent.}
{}
{Measures the REU non-destructively.}
{jsr ultimate\_reu\_size\
sta banks\
stx banks+1}
{16 banks means 1\,MB; 256 banks means 16\,MB.}

\asmref{ultimate\_reu\_stash}
{\cmd{ult\_addr} is a C64 address; the low 24 bits of \cmd{ult\_reu} are the
REU address and its high byte is ignored;
the low word of \cmd{ult\_reulen} is the 16-bit length and zero means 65536
bytes; its high word is ignored.}
{\cmd{A} is a result code after DMA completes.}
{}
{Copies C64 memory into the REU.}
{setword ult\_addr,\$0400\
setdword ult\_reu,0\
setdword ult\_reulen,1000\
jsr ultimate\_reu\_stash}
{The text screen is copied to REU address zero.}

\asmref{ultimate\_reu\_fetch}
{\cmd{ult\_addr}, \cmd{ult\_reu} and \cmd{ult\_reulen} describe the same
ranges as \cmd{ultimate\_reu\_stash}.}
{\cmd{A} is a result code after DMA completes.}
{}
{Copies REU data into C64 memory.}
{setword ult\_addr,\$0400\
jsr ultimate\_reu\_fetch}
{The selected REU bytes replace C64 memory.}

\asmref{ultimate\_reu\_load}
{A readable file is open; \cmd{ult\_reu} is destination and
\cmd{ult\_reulen} is a 32-bit length.}
{\cmd{A} is a result code.}
{}
{Transfers the open file directly into expansion memory.}
{setdword ult\_reu,0\
setdword ult\_reulen,65536\
jsr ultimate\_reu\_load}
{The bytes do not pass through C64 RAM.}

\asmref{ultimate\_reu\_save}
{A writable file is open; \cmd{ult\_reu} is source and
\cmd{ult\_reulen} is a 32-bit length.}
{\cmd{A} is a result code.}
{}
{Transfers expansion memory directly into the open file.}
{setdword ult\_reu,0\
setdword ult\_reulen,65536\
jsr ultimate\_reu\_save}
{The bytes do not pass through C64 RAM.}

\begin{tip}
Use REU memory for large level data, decompression sources, sampled audio and
screen backups. Stash/fetch swap live C64 memory; load/save avoid routing bulk
file data through the C64 address space.
\end{tip}

\asmref{ultimate\_audio\_init}

{\cmd{A} is a result code.}
{}
{Probes the PCM engine with a silent byte.}
{jsr ultimate\_audio\_init\

\asmref{ultimate\_audio\_available}

{\cmd{A} is 1 after successful audio initialization, otherwise 0.}
{}
{Reads cached audio state.}
{jsr ultimate\_audio\_available}

\asmref{ultimate\_audio\_version}

{\cmd{A} is the raw module-version byte.}
{}
{Reads the mapped engine's version register.}
{jsr ultimate\_audio\_version\
sta version}
{\cmd{version} is valid only after audio initialization.}

\asmref{ultimate\_audio\_configure}
{\cmd{A/X} points to a \cmd{UA\_VOICE\_SIZE} descriptor. Named offsets hold
channel 0--6; only \cmd{UA\_CTRL\_FLAGS}; volume 0--63; pan 0--15; a 24-bit REU
start and nonzero length that do not pass 16\,MB; repeat offsets; and a nonzero
rate. A 16-bit or interleaved length is even. Repeat requires
\cmd{repeat\_a < repeat\_b <= length}.}
{\cmd{A} is a result code; the channel is configured but stopped.}
{}
{Validates and programs one PCM voice.}
{lda \#0:sta voice+UA\_VOICE\_FLAGS\
lda \#<voice:ldx \#>voice:jsr ultimate\_audio\_configure\
lda \#UA\_CTRL\_16BIT:sta voice+UA\_VOICE\_FLAGS\
lda \#<voice:ldx \#>voice:jsr ultimate\_audio\_configure\
lda \#UA\_CTRL\_INTERLEAVE:sta voice+UA\_VOICE\_FLAGS\
lda \#<voice:ldx \#>voice:jsr ultimate\_audio\_configure\
lda \#UA\_CTRL\_REPEAT:sta voice+UA\_VOICE\_FLAGS\
lda \#<voice:ldx \#>voice:jsr ultimate\_audio\_configure\
lda \#UA\_CTRL\_IRQ:sta voice+UA\_VOICE\_FLAGS\
lda \#<voice:ldx \#>voice:jsr ultimate\_audio\_configure}
{The examples isolate 8-bit mono, 16-bit, interleave, repeat and IRQ; valid

\asmref{ultimate\_audio\_stop}
{\cmd{A} is channel 0--6.}
{\cmd{A} is a result code.}
{}
{Closes one channel's gate.}
{lda \#0\
jsr ultimate\_audio\_stop}
{Channel zero stops.}

\asmref{ultimate\_audio\_irq\_clear}

\section{HTTP Request Bodies}

Body values are built inside the Ultimate. For keyed JSON/object and URL-form
values, \cmd{ult\_url} points to an ASCII key of at most
\cmd{ULTIMATE\_HTTP\_KEY\_MAX} bytes. For an array element, point it to a
one-byte NUL string; a null pointer is invalid.

\asmref{ultimate\_http\_body}
{\cmd{A} is \cmd{HTTP\_BODY\_JSON\_OBJECT},
\cmd{HTTP\_BODY\_JSON\_ARRAY}, \cmd{HTTP\_BODY\_URL\_ENCODED}, or
\cmd{HTTP\_BODY\_BINARY}.}
{\cmd{A} is a result code; \cmd{ult\_body} is a handle or \$FF on failure.}
{}
{Allocates an empty firmware-side request body.}
{lda \#HTTP\_BODY\_BINARY:jsr ultimate\_http\_body\
lda \#HTTP\_BODY\_JSON\_OBJECT:jsr ultimate\_http\_body\
lda \#HTTP\_BODY\_JSON\_ARRAY:jsr ultimate\_http\_body\
lda \#HTTP\_BODY\_URL\_ENCODED:jsr ultimate\_http\_body}
{The calls create binary, JSON-object, JSON-array and URL-encoded bodies; free
each successful \cmd{ult\_body} handle.}

\asmref{ultimate\_http\_body\_free}
{\cmd{ult\_body} is a body handle.}
{\cmd{A} is a result code.}
{}
{Releases one firmware-side body slot.}
{jsr ultimate\_http\_body\_free}
{The handle is no longer usable.}

\asmref{ultimate\_http\_body\_clear}
{\cmd{ult\_body} is a body handle.}
{\cmd{A} is a result code.}
{}
{Removes all values but preserves the body format and slot.}
{jsr ultimate\_http\_body\_clear}
{The body is empty and reusable.}

\asmref{ultimate\_http\_body\_up}
{\cmd{ult\_body} is positioned inside a nested object or array.}
{\cmd{A} is a result code.}
{}
{Moves the body cursor to its parent container.}
{jsr ultimate\_http\_body\_up}

\asmref{ultimate\_http\_body\_int}

\asmref{ultimate\_http\_body\_bool}

\asmref{ultimate\_http\_body\_string}

\asmref{ultimate\_http\_body\_object}

\section{Low-Level UCI Transport}

Use these entries for commands that the service layer does not wrap. The
\cmd{uci\_request} layout and \cmd{UCI\_REQ\_*} offsets are generated in
\cmd{uci\_protocol.inc}. A request's argument, payload, data and status spans
remain caller-owned. \cmd{uci\_more} is the exported byte read after a
boundary-preserving call; nonzero means another reply block follows.

\asmref{uci\_present}
{None.}
{\cmd{A} is 1 when the documented signature is present, otherwise 0.}
{}
{Performs a fast register-only presence probe.}
{jsr uci\_present\
beq absent}
{This does not prove that the interface can finish a command.}

\asmref{uci\_ident}
{None.}
{\cmd{A} is the raw identification register.}
{}
{Reads diagnostic identity without blocking.}
{jsr uci\_ident\
sta ident}
{\cmd{ident} contains the unvalidated register byte.}

\asmref{uci\_req\_clear}
{None.}
{No defined register result; all \cmd{UCI\_REQ\_SIZE} bytes of
\cmd{uci\_req} are zero.}
{}
{Clears the SDK-owned request block before selected fields are filled.}
{jsr uci\_req\_clear}
{No stale pointer or length remains in \cmd{uci\_req}.}

\asmref{uci\_exec\_block}
{Fill the exported \cmd{uci\_req} block.}
{\cmd{A} is a result code; output lengths are written into the block.}
{}
{Runs the SDK-owned request block to completion.}
{jsr uci\_req\_clear\
lda \#UCI\_TARGET\_DOS1\
sta uci\_req+UCI\_REQ\_TARGET\
jsr uci\_exec\_block}
{The request completes and all reply blocks are drained.}

\asmref{uci\_exec}
{\cmd{A/X} points to any writable \cmd{UCI\_REQ\_SIZE} request block.}
{\cmd{A} is a result code; output lengths are written into that block.}
{}
{Runs a caller-owned request block to completion.}
{lda \#<request\
ldx \#>request\
jsr uci\_exec}
{The caller may keep several independent request blocks.}

\asmref{uci\_exec\_first}
{\cmd{A/X} points to a writable request block.}
{\cmd{A} is a result for the first reply block; \cmd{uci\_more} describes
whether another follows.}
{}
{Starts a boundary-preserving exchange such as a directory walk.}
{lda \#<request\
ldx \#>request\
jsr uci\_exec\_first}
{Only the first logical reply block is retained.}

\asmref{uci\_exec\_next}
{A boundary-preserving exchange is live and \cmd{uci\_more} is nonzero.}
{\cmd{A} is the next block's result; \cmd{uci\_more} is updated.}
{}
{Advances the live exchange by one reply block.}
{lda uci\_more\
beq finished\
jsr uci\_exec\_next}
{The request block's data length is reset for this block.}

\asmref{uci\_abort}
{None.}
{\cmd{A} is a result code.}
{}
{Abandons any live exchange and returns the interface to idle.}
{jsr uci\_abort}
{A stopped directory walk no longer holds the interface.}

\asmref{uci\_set\_timeout\_a}
{\cmd{A} is a budget in units of 256 polls; zero means forever.}
{No defined register result.}
{}
{Sets the transport polling budget. Wall time varies with CPU speed.}
{lda \#UCI\_TIMEOUT\_DEFAULT\
jsr uci\_set\_timeout\_a}
{Later bounded calls use the default budget.}

\asmref{uci\_get\_timeout\_a}
{None.}
{\cmd{A} is the current timeout budget; zero means forever.}
{}
{Reads the current polling budget.}
{jsr uci\_get\_timeout\_a\
sta saved\_timeout}
{The byte can restore caller policy after a temporary change.}

\asmref{uci\_last\_code}
{None.}
{\cmd{A/X} is the most recently decoded raw device status, or \$FFFF when
none was supplied.}
{}
{Returns target-specific diagnostic detail.}
{jsr uci\_last\_code\
sta device\_code\
stx device\_code+1}
{Interpret the number only in the context of the target just called.}
